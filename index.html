/** * @license * SPDX-License-Identifier: Apache-2.0 */ import { Category, Product, MainConfig } from '../types'; export function generateStandaloneHtml( categories: Category[], products: Product[], config: MainConfig ): string { const visibleCategories = categories.filter(c => c.isVisible); const visibleProducts = products.filter(p => p.isVisible); // Generate original static servers content + support custom items as well const cardsHtml = visibleProducts.map(p => { const isRed = p.logoBg.includes('b91c1c') || p.id.toLowerCase().includes('red'); const accentColor = isRed ? '#ff3333' : '#6366f1'; const accentBorder = isRed ? 'rgba(255, 51, 51, 0.25)' : 'rgba(99, 102, 241, 0.25)'; return `
${p.logoText || 'TV'}

${p.name}

${p.description}

فتح السيرفر والتفاصيل
`; }).join('\n'); // Generate Panels / Modals for each product const panelsHtml = visibleProducts.map(p => { const isRed = p.logoBg.includes('b91c1c') || p.id.toLowerCase().includes('red'); const accentColor = isRed ? '#ff3333' : '#6366f1'; const accentBorder = isRed ? 'rgba(255, 51, 51, 0.3)' : 'rgba(99, 102, 241, 0.3)'; const btnBg = isRed ? '#ff1111' : '#4f46e5'; // Stats boxes HTML const statsHtml = p.stats.map(s => `

${s.value}

${s.label}

`).join(''); // Downloads/Instructions HTML const downloadsHtml = p.downloads.map(dl => { if (dl.type === 'download' && dl.url) { return `

${dl.title}

${dl.content}

اضغط هنا للتحميل المباشر
`; } else if (dl.type === 'code') { return `

${dl.title}

${dl.content}
${dl.subtext ? `

${dl.subtext}

` : ''}
`; } else if (dl.type === 'text') { const lines = dl.content.split('\n').join('
'); return `

${dl.title}

${lines}

${dl.subtext ? `

${dl.subtext}

` : ''}
`; } else { const lines = dl.content.split('\n').map(l => `• ${l}`).join('
'); return `

${dl.title}

${lines}

`; } }).join(''); // Price items HTML const pricesHtml = p.prices.map(pr => `

${pr.duration}

${pr.price}
`).join(''); return `

${p.name}

${p.description}

${statsHtml}
${downloadsHtml}
باقات أسعار الاشتراك المتاحة
${pricesHtml}

${p.noticeTitle}

${p.noticeBody}

`; }).join('\n'); // Render categories filters selectors const categoriesFilterHtml = visibleCategories.map((c, index) => { return ``; }).join('\n'); const policyTextConverted = config.usePolicy.split('\n').map(line => line.startsWith('-') ? `• ${line.substring(2)}` : line).join('

'); return ` ${config.storeName} | أفضل سيرفرات IPTV والمنتجات الرقمية في العراق
IQTV
4K

${config.storeName}

أسرع وأضمن خدمة بث في العراق 🇮🇶

الاشتراكات والأسعار سياسة الاستخدام

${config.heroTitle}

${config.heroSubtitle}

طلب تجربة مجانية 12 ساعة استعرض الاشتراكات

اقسام المتجر والاشتراكات

اختر السيرفر أو الباقة التي تناسبك واضغط حجز لإجراء طلب الحجز الفوري وإصدار الفاتورة

${categoriesFilterHtml}
${cardsHtml}
${panelsHtml}
×

سياسة الاستخدام والخصوصية 🔐

${policyTextConverted}

×

تفاصيل حجز الاشتراك

IQTV
4K

${config.storeName} — الوكيل المعتمد لجميع السيرفرات الأصلية في العراق 🏆

سياسة الاستخدام واتساب الدعم
`; }
